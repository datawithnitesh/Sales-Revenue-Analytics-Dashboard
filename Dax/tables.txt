1. Calendar = 
   VAR MinDate = DATE(
        YEAR(MIN(vw_SalesDashboard_Enhanced[OrderDate])),
        1,
1)
VAR MaxDate = DATE(
        YEAR(MAX(vw_SalesDashboard_Enhanced[OrderDate])),
        12,
        31 )
RETURN
ADDCOLUMNS(
    CALENDAR(MinDate, MaxDate),
    //-------------------------
    // Date Keys
    //-------------------------
    "Date Key", FORMAT([Date], "YYYYMMDD"),
    //-------------------------
    // Year
    //-------------------------
    "Year", YEAR([Date]),
    "Year Short",
        RIGHT(YEAR([Date]),2),
    //-------------------------
    // Quarter
    //-------------------------
    "Quarter Number",
        QUARTER([Date]),
    "Quarter",
        "Q" & QUARTER([Date]),
    //-------------------------
    // Month
    //-------------------------
    "Month Number",
        MONTH([Date]),

    "Month",
        FORMAT([Date],"MMMM"),

    "Month Short",
        FORMAT([Date],"MMM"),
    //-------------------------
    // Year Month
    //-------------------------
    "Year Month",
       FORMAT([Date],"YYYY-MMM"),
    "Year Month Sort",
        YEAR([Date])*100 + MONTH([Date]),
    //-------------------------
    // Week
    //-------------------------
    "Week Number",
        WEEKNUM([Date],2),
    //-------------------------
    // Day
    //-------------------------
    "Day",
        DAY([Date]),
    "Day Name",
        FORMAT([Date],"dddd"),
    "Day Short",
        FORMAT([Date],"ddd"),
    "Day Of Week",
        WEEKDAY([Date],2),
    //-------------------------
    // Weekend
    //-------------------------
    "Weekend",
        IF(WEEKDAY([Date],2)>=6,"Weekend","Weekday"),
    //-------------------------
    // Fiscal (April Start)
    //-------------------------
    "Fiscal Year",
        "FY" &
        IF(
            MONTH([Date])>=4,
            YEAR([Date])+1,
            YEAR([Date])
        ),
    "Fiscal Month",
        MOD(MONTH([Date])+8,12)+1
)

    }
